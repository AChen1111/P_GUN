local PlayerBullet = CS.Game.Gameplay.PlayerBullet
local Vector2 = CS.UnityEngine.Vector2

xlua.hotfix(PlayerBullet,"Init",function (self,shootDir, bulletDamage, bulletSpeed)
    self.dir = Vector2(shootDir.x,shootDir.y)--更改射击方向,展示热更新功能
    self.damage = bulletDamage
    self.speed = bulletSpeed
end)

-- 热修拥有其静态回调的释放职责.
return function()
    xlua.hotfix(PlayerBullet, "Init", nil)
end
