Buff 系统

玩家身上的临时效果统一管理

Buff 可以理解成挂在玩家身上的临时规则, 比如加速, 中毒, 加最大生命或净化负面状态. 玩家获得 Buff 后, `BuffManager` 负责记录剩余时间和层数, 普通数值直接按配置重算, 需要特殊逻辑时再交给 Lua 脚本处理.

  • 获得. 通过 Buff id 从 `BuffDataBase` 找配置, 不在技能或物品代码里硬写数值.
  • 运行. `BuffManager` 每帧扣时间, 到期自动移除, 重复获得时重置时间或叠层.
  • 表现. 属性变化后通知 `Player` 和 Buff 状态栏刷新, UI 不自己维护一份状态.
BuffDataBase 配置表 BuffManager 玩家管理器 剩余时间 / 层数 属性重算 Lua 特殊回调 Player / Buff UI 刷新
BuffManager.cs

相关代码: 添加 Buff 与属性计算

重复获得 Buff 时只改运行时状态, 属性最终值由所有 Buff 的配置统一计算.

Assets/Scripts/Gameplay/Buffs/Core/BuffManager.cs

public BuffRuntimeInfo AddBuff(Buff buff, UnityEngine.Object source)
{
    if (buff == null) return null;

    var previousMaxHp = GetOwnerMaxHp();
    if (buffInfoMap.TryGetValue(buff.Id, out var existing))
    {
        if (buff.IsPermanent)
        {
            // 永久 Buff 重复获得时只增加层数, 保留 Lua 运行时状态.
            existing.Source = source;
            existing.StackCount += 1;
            existing.IsPermanent = true;
        }
        else
        {
            ResetBuffRuntimeInfo(existing, buff, source);
        }

        TriggerOnAdd(existing);
        NotifyOwnerStatsChanged(previousMaxHp);
        NotifyBuffsChanged();
        return existing;
    }

    var info = CreateBuffRuntimeInfo(buff, source);
    if (info == null) return null;

    info.Index = buffs.Count;
    buffs.Add(info);
    buffInfoMap[buff.Id] = info;
    TriggerOnAdd(info);
    NotifyOwnerStatsChanged(previousMaxHp);
    NotifyBuffsChanged();
    return info;
}

public float CalculateStat(StatType statType, float baseValue)
{
    var flat = 0f;
    var percentAdd = 0f;
    var finalMul = 1f;

    for (var i = 0; i < buffs.Count; i++)
    {
        var stackCount = Mathf.Max(1, buffs[i].StackCount);
        var modifiers = buffs[i].Buff.Modifiers;
        for (var j = 0; j < modifiers.Count; j++)
        {
            var modifier = modifiers[j];
            if (modifier == null || modifier.StatType != statType) continue;

            // 同一属性按固定值, 百分比, 最终倍率三个分区累计.
            switch (modifier.ModifierType)
            {
                case ModifierType.Flat:
                    flat += modifier.Value * stackCount;
                    break;
                case ModifierType.PercentAdd:
                    percentAdd += modifier.Value * stackCount;
                    break;
                case ModifierType.FinalMul:
                    for (var stackIndex = 0; stackIndex < stackCount; stackIndex++)
                    {
                        finalMul *= modifier.Value;
                    }
                    break;
            }
        }
    }

    return (baseValue + flat) * (1f + percentAdd) * finalMul;
}

Runtime database loading

DataBaseManager 并发加载全局数据库

进入战斗或读档前确保 `ItemDatabase`, `WeaponDatabase`, `BuffDataBase`, `EnemyDatabase` 都已加载.

Assets/Scripts/Gameplay/Managers/DataBaseManager.cs

public async Task EnsureLoadedAsync()
{
    if (IsLoaded)
    {
        return;
    }

    if (loadingTask != null)
    {
        await loadingTask;
        return;
    }

    loadingTask = LoadAllInternalAsync();
    try
    {
        await loadingTask;
    }
    finally
    {
        loadingTask = null;
    }
}

private async Task LoadAllInternalAsync()
{
    IsLoaded = false;
    var loader = AddressableLoader.Instance;
    if (loader == null)
    {
        throw new InvalidOperationException($"{nameof(AddressableLoader)} must exist before loading databases.");
    }

    var itemTask = loader.LoadAssetAsync<ItemDatabase>(itemDatabaseKey);
    var weaponTask = loader.LoadAssetAsync<WeaponDatabase>(weaponDatabaseKey);
    var buffTask = loader.LoadAssetAsync<BuffDataBase>(buffDataBaseKey);
    var enemyTask = loader.LoadAssetAsync<EnemyDatabase>(enemyDatabaseKey);

    await Task.WhenAll(itemTask, weaponTask, buffTask, enemyTask);
}

Addressables wrapper

AddressableLoader 缓存与并发请求合并

同一 address 的并发请求共享同一个加载任务, 并对类型不匹配和加载失败做明确报错.

Assets/Scripts/Core/HotUpdate/AddressableLoader.cs

public async Task<TAsset> LoadAssetAsync<TAsset>(string address)
    where TAsset : UnityEngine.Object
{
    if (string.IsNullOrWhiteSpace(address))
    {
        throw new ArgumentException("Address must not be empty.", nameof(address));
    }

    if (TryGetCachedAsset(address, out TAsset cachedAsset))
    {
        return cachedAsset;
    }

    if (loadingTasksByAddress.TryGetValue(address, out var existingTask))
    {
        return CastLoadedAsset<TAsset>(address, await existingTask);
    }

    var task = LoadAssetInternalAsync<TAsset>(address);
    loadingTasksByAddress[address] = task;
    try
    {
        return CastLoadedAsset<TAsset>(address, await task);
    }
    finally
    {
        loadingTasksByAddress.Remove(address);
    }
}
xLua startup entry

LuaManager 执行启动热修

先预加载 `hotfix` 标签下所有 Lua 文本, 再执行 `hotfix/main`, 让 xLua `require` 只读同步缓存.

Assets/Scripts/Gameplay/Lua/LuaManager.cs

public async Task ExecuteStartupHotfixAsync()
{
    if (startupHotfixExecuted) return;

    var loader = AddressableLoader.Instance;
    if (loader == null)
    {
        throw new InvalidOperationException($"{nameof(AddressableLoader)} must exist before executing startup hotfix.");
    }

    await PreloadHotfixLuaModulesAsync(loader);
    var hotfixEntry = await loader.LoadAssetAsync<TextAsset>(StartupHotfixAddress);
    if (hotfixEntry == null)
    {
        throw new InvalidOperationException($"Startup hotfix asset is null. Address: {StartupHotfixAddress}.");
    }

    luaEnv.DoString(hotfixEntry.text, hotfixEntry.name);
    startupHotfixExecuted = true;
}

玩法系统

Inventory use rule

PlayerInventory 使用物品

只有至少一个效果可用时才消耗物品, 避免满血治疗或无负面 Buff 净化造成浪费.

Assets/Scripts/Items/Inventory/PlayerInventory.cs

public bool Use(int itemId)
{
    if (!stacksById.TryGetValue(itemId, out var stack) || stack.Count <= 0)
    {
        return false;
    }

    var ctx = BuildUseContext();
    var usableEffects = stack.GetUsableEffects(ctx);
    if (usableEffects.Count == 0)
    {
        EventCenter.Trigger(CoreEvents.PlayerHeadMessageRequested, new PlayerHeadMessageEvent("当前无法使用", 1.5f));
        return false;
    }

    for (int i = 0; i < usableEffects.Count; i++)
    {
        usableEffects[i].OnPick(ctx);
    }

    stack.TryConsumeOne();
    if (stack.Count <= 0)
    {
        stacksById.Remove(itemId);
        orderedStacks.Remove(stack);
    }

    EventCenter.Trigger(ItemEvents.InventoryChanged);
    return true;
}

Stat formula

BuffManager 统一属性公式

`Flat`, `PercentAdd`, `FinalMul` 三段式累计, 让数值 Buff 不需要写 Lua.

{

            {

                    flat += modifier.Value * stackCount;
                    break;

                    percentAdd += modifier.Value * stackCount;
                    break;

                    }
                    break;

public T Get(T prefab) {
    if(prefab == null) {
        Debug.LogError($"{GetType().Name}.Get failed: prefab is null.", this);
        return null;
    }

    return GetOrCreatePool(prefab).Get();

public void Release(T item) {
    if (item == null || !item.gameObject.activeSelf) return;

    if(_instance2Pool.TryGetValue(item, out var pool)) {
        pool.Release(item);
        return;
    }

    // 如果对象没有登记在任何池里, 说明它不是由本池创建的, 直接销毁避免泄漏.
    Destroy(item.gameObject);
}

public static SaveOperationResult SaveToSlot(int slotIndex)
{
    if (SceneManager.GetActiveScene().name != GameplaySceneName)
    {
        return SaveOperationResult.Fail("只有游戏场景可以保存.");
    }

    if (FightRoom.currentFightRoom != null)
    {
        return SaveOperationResult.Fail("战斗中不能保存.");
    }

    var player = PlayerRegistry.Current;
    if (player == null)
    {
        return SaveOperationResult.Fail("保存失败, 找不到玩家.");
    }

    var data = SaveDataBuilder.Build(player, SaveVersion);
    SaveSlotStorage.WriteSlot(slotIndex, data);
    SaveSlotSnapshotCapture.Capture(slotIndex);
    Debug.Log($"存档完成, Slot: {slotIndex}, Path: {SaveSlotStorage.GetSlotPath(slotIndex)}.");
    return SaveOperationResult.Ok("保存成功.", data);
}

public static async Task<SaveOperationResult> LoadFromSlotAsync(int slotIndex)
{
    var data = SaveSlotStorage.ReadSlot(slotIndex);
    if (data == null)
    {
        return SaveOperationResult.Fail("读取失败, 槽位为空.");
    }

    await EnsureDatabasesLoadedAsync();
    pendingLoadData = data;
    SceneManager.LoadScene(GameplaySceneName);
    return SaveOperationResult.Ok("正在进入游戏场景并恢复存档.", data);
}

public void Push(UIPanelBase panel)
{
    if (panel == null)
    {
        Debug.LogError("UIStackManager入栈失败, 面板不能为空.");
        return;
    }

    PushInternal(panel, true);
}

public bool Pop()
{
    if (panelStack.Count <= 1)
    {
        // 主面板作为栈底, 默认不允许被普通返回关闭.
        return false;
    }

    UIPanelBase topPanel = panelStack.Pop();

    if (topPanel != null)
    {
        HidePanel(topPanel);
    }

    UIPanelBase resumePanel = Peek();

    if (resumePanel != null)
    {
        // 恢复新的栈顶面板交互.
        resumePanel.Resume();
    }

    return true;
}
